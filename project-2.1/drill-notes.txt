---BOILERPLATE- #

terraform {
    required_providers {
        aws = {
            source = "hashicorp/aws"
            version = "~> 5.0"
        }
    }
}

provider "aws" {
    region = "us-east-1"
    profile = "portfolio
}

---BASIC RESOURCE BUCKET BLOCK--- #

resource "aws_s3_bucket" "example_bucket" {
    bucket = "my_bucket"
}

---VARIABLE BLOCK, REQUIRED VS OPTIONAL--- #

variable "role_name" {
    description = "the name of the role"
    type = string
}

variable "department" {
    description = "the department of the role"
    type = string
    default = "general"
}


---OUTPUT BLOCK---

output "role_arn" {
    description = "ARN for the role"
    name = aws_iam_role.abac_role.arn
}

---TRUST POLICY---

resource "aws_iam_role" "abac_role" {
    name = var.role_name

    tag = {
        department = var.department
    }

    assume_role_policy = jsonencode ({
        Version = "2012-10-17"
        Statement = [
            {
            Action = "sts:AssumeRole"
            Effect = "Allow"
            Principal = {
                AWS = "arn:aws:iam::${var.account_id}:root"
                }
            }
        ]
    })
}
variable "account_id" {
  description = "AWS account ID - supply your own, never commit a real one"
  type        = string
}
